cmake_minimum_required(VERSION 3.25)
project(waddle-game-kit)

set(WGK_CONFIG_PROJECT_NAME "NO_NAME_PROVIDED" CACHE STRING "")

set(WGK_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/source")

set(CMAKE_CXX_STANDARD          23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS        OFF)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

file(
	GLOB_RECURSE WGK_SOURCES CONFIGURE_DEPENDS
	"${WGK_SOURCE}/*.cpp"
)

add_executable(${WGK_CONFIG_PROJECT_NAME})

target_sources(
	${WGK_CONFIG_PROJECT_NAME} PRIVATE

	${WGK_SOURCES}
)

target_include_directories(
	${WGK_CONFIG_PROJECT_NAME} PRIVATE

	"${WGK_SOURCE}"
)

target_compile_definitions(
	${WGK_CONFIG_PROJECT_NAME} PRIVATE

	_ALLOW_COMPILER_AND_STL_VERSION_MISMATCH

	WGK_PROJECT_NAME="${WGK_CONFIG_PROJECT_NAME}"
)

install(
	FILES
		"$<TARGET_FILE_DIR:${WGK_CONFIG_PROJECT_NAME}>/index.html"
		"$<TARGET_FILE_DIR:${WGK_CONFIG_PROJECT_NAME}>/$<TARGET_FILE_BASE_NAME:${WGK_CONFIG_PROJECT_NAME}>.js"
		"$<TARGET_FILE_DIR:${WGK_CONFIG_PROJECT_NAME}>/$<TARGET_FILE_BASE_NAME:${WGK_CONFIG_PROJECT_NAME}>.wasm"
	DESTINATION "."
)
